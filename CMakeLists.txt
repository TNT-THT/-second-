cmake_minimum_required(VERSION 3.20)
project(PetDesk VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Portable build: no redistributable installer, unreviewed plugins or DLL search for Cubism.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W3 /permissive- /EHsc /Gy)
else()
  add_compile_options(-Wall -Wextra)
endif()
add_library(pet_rules STATIC engine/rules.cpp engine/express.cpp engine/sha256.cpp engine/scene.cpp engine/motion.cpp)
target_compile_definitions(pet_rules PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(pet_rules PUBLIC engine PRIVATE vendor/json)
add_library(pet_avatar STATIC engine/avatar.cpp)
target_include_directories(pet_avatar PRIVATE vendor/json)
target_link_libraries(pet_avatar PUBLIC pet_rules)
set(CUBISM4_INCLUDE "" CACHE PATH "Optional authorized Cubism 4 Native R7 Core headers")
set(CUBISM4_LIBRARY "" CACHE FILEPATH "Optional authorized Cubism 4 Native Core static library")
if(CUBISM4_INCLUDE OR CUBISM4_LIBRARY)
  if(NOT EXISTS "${CUBISM4_INCLUDE}/Live2DCubismCore.h" OR NOT EXISTS "${CUBISM4_LIBRARY}")
    message(FATAL_ERROR "Need both authorized Core headers and library")
  endif()
  target_compile_definitions(pet_avatar PRIVATE PET_OFFICIAL_CORE=1)
  target_include_directories(pet_avatar PRIVATE "${CUBISM4_INCLUDE}")
  target_link_libraries(pet_avatar PRIVATE "${CUBISM4_LIBRARY}")
else()
  add_library(purism STATIC vendor/purism/core.c)
  set_target_properties(purism PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  target_compile_definitions(purism PUBLIC PSM_COMPAT_VERSION=0x05010000L PSM_NO_STDIO=1 _CRT_SECURE_NO_WARNINGS)
  target_include_directories(purism PUBLIC vendor/purism)
  target_link_libraries(pet_avatar PRIVATE purism)
endif()
enable_testing()
add_executable(rules_test tests/rules.cpp)
target_link_libraries(rules_test PRIVATE pet_rules)
add_test(NAME rules COMMAND rules_test)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_library(pet_vault STATIC native/vault.cpp)
  target_include_directories(pet_vault PUBLIC native)
  target_link_libraries(pet_vault PUBLIC pet_rules)
  add_executable(PetDesk WIN32 native/main.cpp native/media.cpp native/model.cpp native/bridge.cpp native/view.cpp native/environment.cpp native/memo.cpp native/workbench.cpp)
  target_include_directories(PetDesk PRIVATE vendor/json)
  target_link_libraries(PetDesk PRIVATE pet_vault pet_avatar windowscodecs user32 gdi32 gdiplus shell32 ole32 uuid dwmapi wtsapi32 winmm sapi ws2_32 bcrypt imm32 comctl32)
  if(MSVC)
    target_link_libraries(PetDesk PRIVATE runtimeobject windowsapp)
    target_link_options(PetDesk PRIVATE /OPT:REF /OPT:ICF /MANIFEST:EMBED /MANIFESTUAC:NO "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/native/PetDesk.manifest")
  else()
    target_sources(PetDesk PRIVATE native/app.rc)
    target_link_options(PetDesk PRIVATE -municode)
  endif()
  set(WHISPER_CPP_DIR "" CACHE PATH "Optional previously provisioned whisper.cpp 1.7.6 source")
  set(WHISPER_TINY_SHA256 "" CACHE STRING "Trusted fingerprint of the local GGML multilingual tiny weights")
  if(WHISPER_CPP_DIR)
    string(LENGTH "${WHISPER_TINY_SHA256}" hash_len)
    if(NOT hash_len EQUAL 64 OR NOT WHISPER_TINY_SHA256 MATCHES "^[a-f0-9]+$")
      message(FATAL_ERROR "A pinned SHA-256 is required; recording is off without verified weights")
    endif()
    foreach(flag BUILD_SHARED_LIBS WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER GGML_BACKEND_DL GGML_OPENMP GGML_NATIVE)
      set(${flag} OFF CACHE BOOL "Compact offline build" FORCE)
    endforeach()
    add_subdirectory("${WHISPER_CPP_DIR}" whisper EXCLUDE_FROM_ALL)
    target_link_libraries(PetDesk PRIVATE whisper)
    target_compile_definitions(PetDesk PRIVATE PET_WHISPER=1 PET_WHISPER_HASH="${WHISPER_TINY_SHA256}")
  endif()
  option(PET_DEV "Developer-only state HUD and private settings reload" OFF)
  if(PET_DEV)
    target_compile_definitions(PetDesk PRIVATE PET_DEV=1)
    target_link_libraries(PetDesk PRIVATE psapi)
  endif()
  add_executable(vault_test tests/vault.cpp)
  target_link_libraries(vault_test PRIVATE pet_vault)
  add_test(NAME windows_vault COMMAND vault_test)
endif()
add_executable(avatar_test tests/avatar.cpp)
target_link_libraries(avatar_test PRIVATE pet_avatar)
